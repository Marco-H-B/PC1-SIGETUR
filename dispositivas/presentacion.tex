% ----------------- SLIDE 3: PERÚ TOURS Y SUS PROBLEMAS -----------------
\begin{frame}{Caso de Estudio: Agencia PERÚ TOURS}
    \begin{columns}[t]
        \begin{column}{0.50\textwidth}
            \begin{block}{\textbf{Sobre la Empresa}}
                \small
                \begin{itemize}
                    \item \textbf{Paquetes a medida:} Turismo nacional y receptivo (Cusco, Arequipa, Selva Central).
                    \item \textbf{Clientes:} \textbf{Socios} (con beneficios y fidelización) y \textbf{No Socios} (clientes nuevos por registrar).
                \end{itemize}
            \end{block}

            \begin{block}{\textbf{¿Cómo trabajan hoy?}}
                \small
                \begin{itemize}
                    \item \textbf{Canales informales:} Pedidos y consultas por correo, WhatsApp o teléfono.
                    \item \textbf{Armado manual:} Cotizaciones e itinerarios calculados en hojas de Excel.
                \end{itemize}
            \end{block}
        \end{column}
        
        \begin{column}{0.46\textwidth}
            \begin{alertblock}{\textbf{Problemas Detectados}}
                \small
                \begin{itemize}
                    \item \textbf{Demora en cotizar (>48h):} Tardan días en responder un presupuesto al cliente.
                    \item \textbf{Ajustes perdidos:} Las observaciones y cambios que pide el turista se traspapelan en los chats.
                    \item \textbf{Sin métricas:} Gerencia no tiene datos de cuántas cotizaciones se caen ni los motivos de cancelación.
                \end{itemize}
            \end{alertblock}
        \end{column}
    \end{columns}
\end{frame}

% ----------------- SLIDE 4: PLANTEAMIENTO DEL PROBLEMA (RUP) -----------------
\begin{frame}{Planteamiento del Problema (Plantilla RUP)}
    \small
    \begin{table}[H]
    \centering
    \begin{tabularx}{\textwidth}{|l|X|}
    \hline
    \textbf{El problema de} & La gestión manual y fragmentada de solicitudes de viaje, cotizaciones, observaciones y pagos en PERÚ TOURS. \\ \hline
    \textbf{Afecta a} & Clientes (turistas socios y no socios), Agentes Receptivos y Gerencia. \\ \hline
    \textbf{Cuyo impacto es} & 
    $\bullet$ Demoras de más de 48 horas en entregar presupuestos. \newline
    $\bullet$ Pérdida de clientes por falta de seguimiento a sus observaciones. \newline
    $\bullet$ Falta de control gerencial sobre las cotizaciones rechazadas o canceladas. \\ \hline
    \textbf{Una solución exitosa} & Desarrollar \textbf{SIGETUR}, un sistema web para registrar solicitudes, emitir cotizaciones, atender observaciones en línea, procesar pagos y emitir reportes de ventas y cancelaciones. \\ \hline
    \end{tabularx}
    \end{table}
\end{frame}

    \begin{block}{\textbf{Propuesta de Valor de SIGETUR}}
        \small
        \begin{tabularx}{\textwidth}{lX}
            \textbf{Para:} & PERÚ TOURS y sus clientes (turistas nacionales y extranjeros). \\
            \textbf{Que necesitan:} & Solicitar paquetes a medida, afiliarse como socios y pagar en línea. \\
            \textbf{El producto:} & \textbf{SIGETUR}, plataforma web de gestión turística. \\
            \textbf{Permite:} & Cotizar itinerarios personalizados, negociar observaciones en línea y procesar cobros. \\
            \textbf{A diferencia de:} & Portales masivos (Despegar, Booking) con paquetes cerrados, y la atención manual por WhatsApp o Excel. \\
            \textbf{Nuestro producto:} & Permite adaptar el viaje al cliente, registrar cada cambio formalmente y darle control de ventas a gerencia.
        \end{tabularx}
    \end{block}
\end{frame}

\begin{frame}{Modelo del Negocio: De los Procesos al Software}
    \begin{block}{\textbf{Transición Metodológica (García Molina et al.)}}
        \small
        Para definir los requisitos del sistema, primero debemos entender y ordenar los procesos de PERÚ TOURS, identificando cuellos de botella y metas cuantificables.
    \end{block}
    \vspace{0.2cm}
    \begin{columns}[t]
        \begin{column}{0.50\textwidth}
            \begin{exampleblock}{\textbf{Objetivos del Negocio (ON)}}
                \small
                \begin{itemize}
                    \item \textbf{ON-01 (Rapidez):} Reducir tiempo de cotización a menos de 4 horas (actualmente $> 48$h).
                    \item \textbf{ON-02 (Crecimiento):} Incrementar afiliación de socios en un 40\%.
                    \item \textbf{ON-03 (Ventas):} Recuperar el 25\% de cotizaciones observadas.
                    \item \textbf{ON-04 (Control):} Monitorear el 100\% de cotizaciones canceladas y sus causas.
                \end{itemize}
            \end{exampleblock}
        \end{column}
        
        \begin{column}{0.46\textwidth}
            \begin{exampleblock}{\textbf{Elementos del Modelo (RUP)}}
                \small
                \begin{itemize}
                    \item \textbf{Actores del Negocio:} Entidades externas que reciben valor de la agencia (Cliente Turista, Pasarela Bancaria).
                    \item \textbf{Trabajadores del Negocio:} Personal interno que ejecuta los procesos (Agente Receptivo, Gerente).
                    \item \textbf{Casos de Uso del Negocio (CUN):} Procesos clave que satisfacen los objetivos de la empresa.
                \end{itemize}
            \end{exampleblock}
        \end{column}
    \end{columns}
\end{frame}

% ----------------- SLIDE 12: OBJETOS DEL NEGOCIO Y CICLO DE ESTADOS -----------------
\begin{frame}{Ciclo de Vida del Objeto Central: Cotización}
    \begin{block}{\textbf{Máquina de Estados (\texttt{c: Cotización})}}
        \small \centering
        \textbf{Camino Principal:} \texttt{[Emitida]} $\longrightarrow$ \texttt{[Aprobada]} $\longrightarrow$ \texttt{[Pagada]}\\[0.2cm]
        \textbf{Negociación:} \texttt{[Emitida]} $\longrightarrow$ \texttt{[Observada]} $\rightleftharpoons$ \texttt{[Reajustada]} $\longrightarrow$ \texttt{[Aprobada]}\\[0.2cm]
        \textbf{Cierre / Desistimiento:} \texttt{[Emitida / Observada]} $\longrightarrow$ \texttt{[Cancelada]}
    \end{block}
    
    \vspace{0.2cm}
    \begin{columns}[t]
        \begin{column}{0.48\textwidth}
            \begin{exampleblock}{\textbf{Lógica de Negociación}}
                \small
                \begin{itemize}
                    \item Si el cliente tiene discrepancias de precio o ruta, pasa a \textbf{[Observada]}.
                    \item El agente emite una versión \textbf{[Reajustada]} hasta su conformidad.
                \end{itemize}
            \end{exampleblock}
        \end{column}
        
        \begin{column}{0.48\textwidth}
            \begin{alertblock}{\textbf{Reglas Críticas del Negocio}}
                \small
                \begin{itemize}
                    \item \textbf{Control de Pago (RN03):} Solo se permite cobrar a cotizaciones en estado \textbf{[Aprobada]}.
                    \item \textbf{Auditoría (RN05):} Todo paso a \textbf{[Cancelada]} registra motivo para el reporte gerencial.
                \end{itemize}
            \end{alertblock}
        \end{column}
    \end{columns}
\end{frame}

% ----------------- SLIDE 14: REQUISITOS NO FUNCIONALES Y REGLAS DE NEGOCIO -----------------
\begin{frame}{Requisitos No Funcionales (FURPS+) y Reglas de Negocio}
    \begin{columns}[t]
        \begin{column}{0.48\textwidth}
            \begin{block}{\textbf{Calidad del Sistema (Modelo FURPS+)}}
                \scriptsize
                \begin{itemize}
                    \item \textbf{Usabilidad:} Interfaz web responsive (móvil y PC) y navegación ágil en un máximo de 4 pasos.
                    \item \textbf{Rendimiento:} Consultas y generación de reportes con respuesta menor a 1.8 segundos.
                    \item \textbf{Seguridad:} Control de acceso por roles (RBAC), tráfico cifrado en HTTPS y claves con Argon2id.
                    \item \textbf{Confiabilidad:} Disponibilidad del servicio $\ge 99.5\%$ y copias de seguridad automáticas diarias.
                \end{itemize}
            \end{block}
        \end{column}
        
        \begin{column}{0.48\textwidth}
            \begin{alertblock}{\textbf{Reglas del Negocio (RN)}}
                \scriptsize
                \begin{itemize}
                    \item \textbf{RN01 (Condición de Socio):} Solo usuarios con cuenta de socio activa pueden solicitar cotizaciones.
                    \item \textbf{RN02 (Cronología):} La fecha de regreso debe ser posterior a la partida, con anticipación mínima de 48h.
                    \item \textbf{RN03 (Regla de Cobro):} Ningún pago se procesa si la cotización no tiene el estado \textbf{[Aprobada]}.
                    \item \textbf{RN04 (Negociación):} Toda cotización observada debe ser reajustada por el agente antes de aceptarse.
                    \item \textbf{RN05 (Auditoría):} El reporte de cancelaciones es exclusivo para el Agente Receptivo y Gerente.
                \end{itemize}
            \end{alertblock}
        \end{column}
    \end{columns}
\end{frame}
